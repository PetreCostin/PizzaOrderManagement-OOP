cmake_minimum_required(VERSION 3.16)
project(PizzaOrderManagementOOP VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pizza_core src/Pizza.cpp src/Order.cpp)
target_include_directories(pizza_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(pizza_app src/main.cpp)
target_link_libraries(pizza_app PRIVATE pizza_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(pizza_tests tests/test_order.cpp)
    target_link_libraries(pizza_tests PRIVATE pizza_core)
    add_test(NAME pizza_tests COMMAND pizza_tests)
endif()
